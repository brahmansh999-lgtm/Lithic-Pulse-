// On-site massing diagram — same stepped contour-blob rendering as
// MassingZoningPreview, but clipped to the REAL site polygon (like
// ZoningHeatmap does for the Zoning step) instead of a synthetic dashed
// boundary, so the architect sees each massing candidate sitting in its
// actual plot on the live map rather than as an abstract floating card.

const CATEGORY_COLOR = {
  units:     { r: 168, g: 143, b: 194 }, // violet
  amenity:   { r: 201, g: 142, b: 164 }, // rose
  green:     { r: 110, g: 170, b: 130 }, // sage
  connector: { r: 207, g: 126, b: 99  }, // terracotta
  parking:   { r: 120, g: 118, b: 136 }, // slate
};

const CATEGORY_LABEL = {
  units: 'Units', amenity: 'Amenity', green: 'Green', connector: 'Connector', parking: 'Parking',
};

const RING_FRACS = [1, 0.82, 0.64, 0.47, 0.31, 0.17];
const RING_ALPHA = [0.16, 0.24, 0.34, 0.46, 0.6, 0.8];
const VERTS = 16;

function seededJitter(seed) {
  let a = seed >>> 0;
  return () => {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function blobPath(cx, cy, r, seed) {
  const rng = seededJitter(seed);
  const pts = [];
  for (let i = 0; i < VERTS; i++) {
    const angle = (i / VERTS) * Math.PI * 2;
    const wobble = 0.78 + rng() * 0.44;
    pts.push([cx + Math.cos(angle) * r * wobble, cy + Math.sin(angle) * r * wobble]);
  }
  let d = `M ${(pts[0][0] + pts[pts.length - 1][0]) / 2} ${(pts[0][1] + pts[pts.length - 1][1]) / 2} `;
  for (let i = 0; i < pts.length; i++) {
    const next = pts[(i + 1) % pts.length];
    const mid = [(pts[i][0] + next[0]) / 2, (pts[i][1] + next[1]) / 2];
    d += `Q ${pts[i][0]} ${pts[i][1]} ${mid[0]} ${mid[1]} `;
  }
  return d + 'Z';
}

function ContourBlob({ x, y, peakR, color, seed }) {
  return (
    <>
      {RING_FRACS.map((frac, i) => (
        <path
          key={i}
          d={blobPath(x, y, peakR * frac, seed * 31 + i * 7)}
          fill={`rgba(${color.r},${color.g},${color.b},${RING_ALPHA[i]})`}
        />
      ))}
    </>
  );
}

function categoryCentroids(zoningFloors) {
  const agg = {};
  for (const floor of zoningFloors || []) {
    for (const b of floor.blocks) {
      const cx = b.x + b.w / 2, cy = b.z + b.d / 2;
      const area = b.w * b.d;
      const a = (agg[b.category] ??= { sx: 0, sy: 0, area: 0 });
      a.sx += cx * area; a.sy += cy * area; a.area += area;
    }
  }
  return Object.entries(agg).map(([category, a]) => ({
    category,
    x: (a.sx / a.area) * 100,
    y: (a.sy / a.area) * 100,
    peakR: Math.max(9, Math.min(30, Math.sqrt(a.area / (zoningFloors.length || 1)) * 60)),
  }));
}

export default function MassingOnSite({ zoningFloors, seed = 1, clipPolygon }) {
  const blobs = useMemo(() => categoryCentroids(zoningFloors), [zoningFloors]);
  if (!blobs.length) return null;

  const clipId = `mos-site-clip-${seed}`;
  const clipPoints = clipPolygon?.length >= 3
    ? clipPolygon.map(p => `${p.x},${p.y}`).join(' ')
    : null;

  return (
    <svg viewBox="0 0 100 100" preserveAspectRatio="none" style={{ width: '100%', height: '100%', overflow: 'visible' }}>
      <defs>
        {clipPoints && (
          <clipPath id={clipId} clipPathUnits="userSpaceOnUse">
            <polygon points={clipPoints} />
          </clipPath>
        )}
      </defs>
      <g clipPath={clipPoints ? `url(#${clipId})` : undefined}>
        {blobs.map(b => (
          <ContourBlob key={b.category} x={b.x} y={b.y} peakR={b.peakR} color={CATEGORY_COLOR[b.category] || CATEGORY_COLOR.units} seed={seed} />
        ))}
        {blobs.map(b => (
          <g key={`label-${b.category}`}>
            <circle cx={b.x} cy={b.y} r="0.9" fill="#fff" />
            <text
              x={b.x} y={b.y - 4}
              textAnchor="middle" dominantBaseline="middle"
              fill="rgba(255,255,255,0.95)"
              fontSize="4.6"
              fontFamily="var(--sans, sans-serif)"
              fontWeight="700"
              stroke="rgba(0,0,0,0.6)"
              strokeWidth="0.3"
              paintOrder="stroke"
            >
              {CATEGORY_LABEL[b.category] || b.category}
            </text>
          </g>
        ))}
      </g>
    </svg>
  );
}

import { useEffect, useMemo, useState } from 'react';
import { generateZoningMassingVariations } from './zoningMassingVariations';

const MASSING_ITERATIONS = 20;

// Step 02b / Massing — the DXF zoning fixes WHERE each category (Units /
// Amenity / Green / Connector / Parking) sits and roughly how big its
// region is; it does NOT fix how many real units, at what density, or
// what massing profile (flat/stepped/cascading/podium) fills the Units
// region — that choice belongs to the generator, and is the actual source
// of variation within one fixed zoning (see zoningMassingVariations.js).
// Every candidate is scored against the 4 real bldg_gl precedents that
// actually fit this site (siteFitPrecedents — not all 11 regardless of
// buildability); the candidate whose best match scores highest is the one
// the pipeline declares as "this is how the real building was reached".
//
// UI: this is now a thin control strip, NOT a full-screen panel — the
// actual massing diagram renders on the live map, on the real site
// polygon (see MassingOnSite + SitePicker's SVGOverlay), one candidate at
// a time, step-through style, so the architect sees how each variation
// actually sits on the real plot instead of comparing abstract floating
// cards. onActiveChange fires the active candidate up to SitePicker so it
// can drive the on-map overlay.
export default function MassingStage({ siteBounds, onBack, onContinue, onActiveChange }) {
  const [index, setIndex] = useState(0);

  const { variations, bestMatch, candidatePrecedents } = useMemo(
    () => generateZoningMassingVariations({ keep: MASSING_ITERATIONS, siteBounds }),
    [siteBounds]
  );

  // Land on the best GLB-converging candidate first, not always index 0 —
  // that's the one the pipeline is actually recommending.
  useEffect(() => {
    if (!bestMatch || !variations.length) return;
    const bestIdx = variations.findIndex(v => v.seed === bestMatch.seed);
    if (bestIdx >= 0) setIndex(bestIdx);
  }, [bestMatch, variations]);

  const active = variations[index] || null;

  useEffect(() => {
    onActiveChange?.(active);
  }, [active, onActiveChange]);

  if (!variations.length) {
    return (
      <div className="msg-strip">
        <button className="msg-back" onClick={onBack}>← Zoning</button>
        <div className="msg-strip-empty">No zoning data available — DXF/READ.dxf extraction may be missing.</div>
      </div>
    );
  }

  const isBestGlbMatch = bestMatch && active?.seed === bestMatch.seed;

  return (
    <div className="msg-strip">
      <button className="msg-back" onClick={onBack}>← Zoning</button>

      <div className="msg-strip-nav">
        <button
          className="msg-nav-btn"
          disabled={index === 0}
          onClick={() => setIndex(i => Math.max(0, i - 1))}
        >
          ←
        </button>
        <div className="msg-strip-label">
          <span className="msg-strip-step">02b / Massing</span>
          <span className="msg-strip-variation">
            {active?.variationLabel} <span className="msg-strip-count">({index + 1} / {variations.length})</span>
            {isBestGlbMatch && <span className="msg-strip-glb-tag">closest to {bestMatch.matchedGLB?.name}</span>}
          </span>
        </div>
        <button
          className="msg-nav-btn"
          disabled={index === variations.length - 1}
          onClick={() => setIndex(i => Math.min(variations.length - 1, i + 1))}
        >
          →
        </button>
      </div>

      {active && (
        <div className="msg-strip-stats">
          <span>{active.plan.totalUnits} homes</span>
          <span>{active.plan.greenPercent}% green</span>
          <span>{active.plan.floors.max} floors</span>
          <span>fit {active.score}</span>
          {active.glbScore != null && <span className="msg-strip-glb-score">{active.glbScore}% match {active.matchedGLB?.name}</span>}
          <span className="msg-strip-precedent-note">scored against {candidatePrecedents.length} site-fitting precedents</span>
        </div>
      )}

      <button
        className="msg-continue"
        disabled={!active}
        onClick={() => active && onContinue?.(active)}
      >
        Generate Building →
      </button>
    </div>
  );
}

function classifyNode(name) {
  const n = (name || '').toLowerCase();
  if (n.startsWith('make2d') || n === 'default' || n === 'layer0') return null; // CAD export noise
  if (n.includes('column')) return 'columns';
  if (n.includes('stair')) return 'stairs';
  if (n.includes('terrace') || n.includes('green')) return 'green';
  if (n.includes('balcon')) return 'green'; // amenity-adjacent, counts toward open/communal space
  if (n.includes('external wall')) return 'externalWalls';
  if (n.includes('glass')) return 'glazing';
  if (n.includes('floor plate') || n === 'slabs' || n.includes('floor')) return 'floors';
  if (n.includes('railing') || n.includes('plinth') || n.includes('lift') || n.includes('corridor')) return 'other';
  return 'other';
}

// World-space AABB for one node's meshes (samples vertices for speed on huge meshes).
function nodeWorldBBox(node) {
  const mesh = node.getMesh();
  if (!mesh) return null;
  const m = node.getWorldMatrix();
  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity, minZ = Infinity, maxZ = -Infinity;
  let any = false;
  for (const prim of mesh.listPrimitives()) {
    const pos = prim.getAttribute('POSITION');
    if (!pos) continue;
    const count = pos.getCount();
    const stride = Math.max(1, Math.floor(count / 500)); // sample, not full read — these meshes are huge
    const v = [0, 0, 0];
    for (let i = 0; i < count; i += stride) {
      pos.getElement(i, v);
      const [x, y, z] = v;
      const wx = m[0]*x + m[4]*y + m[8]*z  + m[12];
      const wy = m[1]*x + m[5]*y + m[9]*z  + m[13];
      const wz = m[2]*x + m[6]*y + m[10]*z + m[14];
      if (wx < minX) minX = wx; if (wx > maxX) maxX = wx;
      if (wy < minY) minY = wy; if (wy > maxY) maxY = wy;
      if (wz < minZ) minZ = wz; if (wz > maxZ) maxZ = wz;
      any = true;
    }
  }
  return any ? { minX, maxX, minY, maxY, minZ, maxZ } : null;
}

// centre spacing). The source GLBs export columns as short per-floor
// SEGMENTS stacked up the building (each ~0.9-1.5m tall), not one mesh per
// full-height column — so the raw leaf list contains the same XZ post
// position repeated at many different Y levels. Comparing all of them
// together pollutes the spacing estimate with duplicate-position noise.
// Fix: only look at segments whose minY sits in the lowest occupied band
// (ground floor posts) — one segment per real column at that level.
function estimateColumnGridM(columnBoxes) {
  if (columnBoxes.length < 4) return null;
  const minYs = columnBoxes.map(b => b.minY).sort((a, b) => a - b);
  const groundLevel = minYs[0];
  const groundBoxes = columnBoxes.filter(b => b.minY - groundLevel < 1.0);
  if (groundBoxes.length < 4) return null;

  // Cluster ground-level centres within 0.5m (snap near-duplicates from
  // segment-splitting/overlap) before measuring spacing.
  const rawCentres = groundBoxes.map(b => [(b.minX + b.maxX) / 2, (b.minZ + b.maxZ) / 2]);
  const centres = [];
  for (const c of rawCentres) {
    if (!centres.some(e => Math.hypot(e[0] - c[0], e[1] - c[1]) < 0.5)) centres.push(c);
  }

  const xs = [...new Set(centres.map(c => Math.round(c[0] * 10) / 10))].sort((a, b) => a - b);
  const zs = [...new Set(centres.map(c => Math.round(c[1] * 10) / 10))].sort((a, b) => a - b);
  const gaps = (arr) => {
    const d = [];
    for (let i = 1; i < arr.length; i++) {
      const g = arr[i] - arr[i - 1];
      if (g > 2.0 && g < 20) d.push(g); // plausible structural span range
    }
    return d;
  };
  const allGaps = [...gaps(xs), ...gaps(zs)];
  if (!allGaps.length) return null;
  allGaps.sort((a, b) => a - b);
  return Math.round(allGaps[Math.floor(allGaps.length / 2)] * 10) / 10; // median gap, 1 decimal
}

// the matched bldg_gl precedent (chosen.matchedGLB.columnPositions,
// extracted by scripts/extractZoning.mjs from the actual GLB geometry),
// mapped onto the chosen massing's real building footprints — not a
// synthetic uniform grid. This is the literal "based on the final GLB
// model" placement the architect asked for: every column you see here
// came from a real measured column in a real finished building, scaled
// onto this massing's footprint(s). Falls back to a uniform grid
// (buildColumnGrid, spec.targetColumnGridM) only when no matched GLB has
// usable column data (e.g. picked precedent's geometry didn't classify
// any "column"-named group). Columns animate in one-by-one ("popping up")
// rather than appearing all at once, so the structural layout reads as a

const COS30 = Math.cos(Math.PI / 6);
const SIN30 = Math.sin(Math.PI / 6);
const project = (x, y, z) => ({ sx: (x - z) * COS30, sy: (x + z) * SIN30 - y });
const pts = (arr) => arr.map(p => `${p.sx.toFixed(2)},${p.sy.toFixed(2)}`).join(' ');

function boxFaces(x0, x1, y0, y1, z0, z1) {
  const p = (x, y, z) => project(x, y, z);
  const c = {
    bnw: p(x0, y0, z0), bne: p(x1, y0, z0), bsw: p(x0, y0, z1), bse: p(x1, y0, z1),
    tnw: p(x0, y1, z0), tne: p(x1, y1, z0), tsw: p(x0, y1, z1), tse: p(x1, y1, z1),
  };
  return {
    top:   [c.tnw, c.tne, c.tse, c.tsw],
    left:  [c.bnw, c.bsw, c.tsw, c.tnw],
    right: [c.bne, c.bse, c.tse, c.tne],
  };
}

const FLOOR_TO_FLOOR_M = 3.2;
const COLUMN_W = 0.45; // visual column cross-section, metres

// Maps the matched GLB's real normalised column positions (0..1 in its OWN
// footprint) onto one building's real-world footprint — same idea as
// scaling a structural drawing onto a different-sized plate. Snapped to
// the building's own grid cells so columns land at sane coordinates.
function realColumnsForBuilding(building, columnPositions) {
  return columnPositions.map(({ x, z }) => ({
    col: building.col + Math.round(x * building.w),
    row: building.row + Math.round(z * building.d),
  }));
}

function buildScene(plan, gridPitchM, matchedGLB) {
  const buildings = (plan?.buildings || []).filter(b => b.units?.length > 0);
  if (!buildings.length) return null;

  const realPositions = matchedGLB?.columnPositions;
  const usingRealGLBColumns = Array.isArray(realPositions) && realPositions.length >= 4;

  const columnsByBuilding = buildings.map(b =>
    usingRealGLBColumns ? realColumnsForBuilding(b, realPositions) : buildColumnGrid(b, gridPitchM)
  );

  const boxes = buildings.map(b => ({
    x0: b.col * CELL, x1: (b.col + b.w) * CELL,
    z0: b.row * CELL, z1: (b.row + b.d) * CELL,
    h: Math.max(1, b.floors) * FLOOR_TO_FLOOR_M,
  }));

  const columns = buildings.flatMap((b, i) => {
    const h = Math.max(1, b.floors) * FLOOR_TO_FLOOR_M;
    return columnsByBuilding[i].map(c => ({ x: c.col * CELL, z: c.row * CELL, h }));
  });

  const allPts = boxes.flatMap(b => {
    const f = boxFaces(b.x0, b.x1, 0, b.h, b.z0, b.z1);
    return [...f.top, ...f.left, ...f.right];
  });
  const xs = allPts.map(p => p.sx), ys = allPts.map(p => p.sy);
  const minX = Math.min(...xs), maxX = Math.max(...xs);
  const minY = Math.min(...ys), maxY = Math.max(...ys);
  const pad = Math.max(3, (maxX - minX) * 0.08);
  const viewBox = `${minX - pad} ${minY - pad} ${maxX - minX + pad * 2} ${maxY - minY + pad * 2}`;

  return { boxes, columns, viewBox, usingRealGLBColumns };
}

  const R = 6371000;
  const toRad = d => d * Math.PI / 180;

import { useMemo } from 'react';

// On-site massing diagram — solid isometric boxes per zone category,
// exploded/stacked, matching the architect's "Spatial Zoning Diagram"
// reference exactly (solid coloured boxes, not blurred contour blobs).
// Clipped to the REAL site polygon (same SVGOverlay/clip pattern as
// ZoningHeatmap) so the architect sees the massing sitting in its actual
// plot. Labels sit just outside each box (small leader dot + text), not
// stacked on top of each other in the middle of the diagram.

const COS30 = Math.cos(Math.PI / 6);
const SIN30 = Math.sin(Math.PI / 6);
const project = (x, y, z) => ({ sx: (x - z) * COS30, sy: (x + z) * SIN30 - y });
const pts = (arr) => arr.map(p => `${p.sx.toFixed(2)},${p.sy.toFixed(2)}`).join(' ');

function boxFaces(x0, x1, y0, y1, z0, z1) {
  const p = (x, y, z) => project(x, y, z);
  const c = {
    bnw: p(x0, y0, z0), bne: p(x1, y0, z0), bsw: p(x0, y0, z1), bse: p(x1, y0, z1),
    tnw: p(x0, y1, z0), tne: p(x1, y1, z0), tsw: p(x0, y1, z1), tse: p(x1, y1, z1),
  };
  return {
    top:   [c.tnw, c.tne, c.tse, c.tsw],
    left:  [c.bnw, c.bsw, c.tsw, c.tnw],
    right: [c.bne, c.bse, c.tse, c.tne],
  };
}

const CATEGORY_COLOR = {
  units:     { top: '#cdb7e0', left: '#a98fc2', right: '#bba3d4' }, // lilac
  amenity:   { top: '#e8b9c9', left: '#c98ea4', right: '#dba6ba' }, // rose
  green:     { top: '#f6d9b0', left: '#dcb685', right: '#ecc89c' }, // sand
  connector: { top: '#f0a98c', left: '#cf7e63', right: '#e0937a' }, // coral
  parking:   { top: '#8d86a8', left: '#6c6788', right: '#7c7698' }, // slate
};

const CATEGORY_LABEL = {
  units: 'Units', amenity: 'Amenity', green: 'Green', connector: 'Connector', parking: 'Parking',
};

const FLOOR_H = 0.6;     // per-floor-group slab height, normalised
const EXPLODE_GAP = 0.2; // vertical gap between exploded floor-groups

// Lays out every category block in every floor-group as its own solid box,
// exploded vertically by floor-group (same geometry the original DXF
// extraction produced — see zoningMassingVariations.js's layoutFloor),
// instead of collapsing each category into one blurred blob.
function layoutBoxes(zoningFloors) {
  const items = [];
  let y = 0;
  for (const floor of zoningFloors || []) {
    const y0 = y, y1 = y + FLOOR_H;
    for (const b of floor.blocks) {
      items.push({ x0: b.x, x1: b.x + b.w, z0: b.z, z1: b.z + b.d, y0, y1, category: b.category });
    }
    y = y1 + EXPLODE_GAP;
  }
  return items;
}

// One label per category, anchored to that category's topmost/highest box
// (reads best against the exploded stack) and projected just outside its
// top-right corner — same "leader dot + text beside the box" layout as
// the reference, kept inside the 0-100 viewBox instead of a separate panel.
function categoryLabelAnchors(items) {
  const byCategory = {};
  for (const it of items) {
    const existing = byCategory[it.category];
    if (!existing || it.y1 > existing.y1) byCategory[it.category] = it;
  }
  return Object.entries(byCategory).map(([category, it]) => {
    const corner = project(it.x1, it.y1, it.z0);
    return { category, sx: corner.sx, sy: corner.sy };
  });
}

export default function MassingOnSite({ zoningFloors, seed = 1, clipPolygon }) {
  const items = useMemo(() => layoutBoxes(zoningFloors), [zoningFloors]);
  const labels = useMemo(() => categoryLabelAnchors(items), [items]);
  if (!items.length) return null;

  // Fit the isometric projection into a 0-100 box so it lines up with the
  // clip polygon's own 0-100 coordinate space.
  const allPts = items.flatMap(b => {
    const f = boxFaces(b.x0, b.x1, b.y0, b.y1, b.z0, b.z1);
    return [...f.top, ...f.left, ...f.right];
  });
  const xs = allPts.map(p => p.sx), ys = allPts.map(p => p.sy);
  const minX = Math.min(...xs), maxX = Math.max(...xs);
  const minY = Math.min(...ys), maxY = Math.max(...ys);
  const spanX = (maxX - minX) || 1, spanY = (maxY - minY) || 1;
  const scale = 100 / Math.max(spanX, spanY) * 0.86; // 0.86 leaves room for labels
  const toView = (sx, sy) => ({
    vx: 50 + (sx - (minX + maxX) / 2) * scale,
    vy: 50 + (sy - (minY + maxY) / 2) * scale,
  });

  const clipId = `mos-site-clip-${seed}`;
  const clipPoints = clipPolygon?.length >= 3
    ? clipPolygon.map(p => `${p.x},${p.y}`).join(' ')
    : null;

  return (
    <svg viewBox="0 0 100 100" preserveAspectRatio="none" style={{ width: '100%', height: '100%', overflow: 'visible' }}>
      <defs>
        {clipPoints && (
          <clipPath id={clipId} clipPathUnits="userSpaceOnUse">
            <polygon points={clipPoints} />
          </clipPath>
        )}
      </defs>
      <g clipPath={clipPoints ? `url(#${clipId})` : undefined}>
        {items.map((b, i) => {
          const f = boxFaces(b.x0, b.x1, b.y0, b.y1, b.z0, b.z1);
          const c = CATEGORY_COLOR[b.category] || CATEGORY_COLOR.units;
          const toPts = (face) => pts(face.map(p => { const v = toView(p.sx, p.sy); return { sx: v.vx, sy: v.vy }; }));
          return (
            <g key={i}>
              <polygon points={toPts(f.left)}  fill={c.left}  stroke="#000" strokeWidth="0.15" />
              <polygon points={toPts(f.right)} fill={c.right} stroke="#000" strokeWidth="0.15" />
              <polygon points={toPts(f.top)}   fill={c.top}   stroke="#000" strokeWidth="0.15" />
            </g>
          );
        })}
        {labels.map(({ category, sx, sy }) => {
          const v = toView(sx, sy);
          const c = CATEGORY_COLOR[category] || CATEGORY_COLOR.units;
          return (
            <g key={`label-${category}`}>
              <circle cx={v.vx} cy={v.vy} r="0.7" fill={c.top} stroke="#fff" strokeWidth="0.25" />
              <text
                x={v.vx + 1.6} y={v.vy}
                textAnchor="start" dominantBaseline="middle"
                fill="rgba(255,255,255,0.95)"
                fontSize="3.4"
                fontFamily="var(--sans, sans-serif)"
                fontWeight="600"
                stroke="rgba(0,0,0,0.65)"
                strokeWidth="0.22"
                paintOrder="stroke"
              >
                {CATEGORY_LABEL[category] || category}
              </text>
            </g>
          );
        })}
      </g>
    </svg>
  );
}

// Isometric, exploded stacked-floor-group massing diagram for one zoning
// variation — matches the architect's "Spatial Zoning Diagram" reference:
// solid coloured boxes (not blurred contour blobs), pastel palette only,
// black background. Standalone diagram (NOT overlaid on the live map —
// a flat top-down map and an isometric 3D diagram can't be made to align,
// so this renders on its own canvas, same as the original massing study).

const COS30 = Math.cos(Math.PI / 6);
const SIN30 = Math.sin(Math.PI / 6);
const project = (x, y, z) => ({ sx: (x - z) * COS30, sy: (x + z) * SIN30 - y });
const pts = (arr) => arr.map(p => `${p.sx.toFixed(2)},${p.sy.toFixed(2)}`).join(' ');

function boxFaces(x0, x1, y0, y1, z0, z1) {
  const p = (x, y, z) => project(x, y, z);
  const c = {
    bnw: p(x0, y0, z0), bne: p(x1, y0, z0), bsw: p(x0, y0, z1), bse: p(x1, y0, z1),
    tnw: p(x0, y1, z0), tne: p(x1, y1, z0), tsw: p(x0, y1, z1), tse: p(x1, y1, z1),
  };
  return {
    top:   [c.tnw, c.tne, c.tse, c.tsw],
    left:  [c.bnw, c.bsw, c.tsw, c.tnw],
    right: [c.bne, c.bse, c.tse, c.tne],
  };
}

const EXPLODE_GAP = 0.22;

function layoutScene(zoningFloors) {
  if (!zoningFloors?.length) return null;
  const items = [];
  let y = 0;
  for (const floor of zoningFloors) {
    const y0 = y, y1 = y + FLOOR_H;
    for (const b of floor.blocks) {
      items.push({ x0: b.x, x1: b.x + b.w, z0: b.z, z1: b.z + b.d, y0, y1, category: b.category });
    }
    y = y1 + EXPLODE_GAP;
  }

  const allPts = items.flatMap(b => {
    const f = boxFaces(b.x0, b.x1, b.y0, b.y1, b.z0, b.z1);
    return [...f.top, ...f.left, ...f.right];
  });
  const xs = allPts.map(p => p.sx), ys = allPts.map(p => p.sy);
  const minX = Math.min(...xs), maxX = Math.max(...xs);
  const minY = Math.min(...ys), maxY = Math.max(...ys);
  const pad = Math.max(0.08, (maxX - minX) * 0.1);
  const viewBox = `${minX - pad} ${minY - pad} ${maxX - minX + pad * 2} ${maxY - minY + pad * 2}`;

  return { items, viewBox };
}

    tnw: p(x0, y1, z0), tne: p(x1, y1, z0), tsw: p(x0, y1, z1), tse: p(x1, y1, z1),
  };
  return { top: [c.tnw,c.tne,c.tse,c.tsw], left: [c.bnw,c.bsw,c.tsw,c.tnw], right: [c.bne,c.bse,c.tse,c.tne] };
}

// shoelace signed area - negative/positive tells winding direction
function signedArea(pts) {
  let a = 0;
  for (let i = 0; i < pts.length; i++) {
    const p1 = pts[i], p2 = pts[(i+1)%pts.length];
    a += p1.sx * p2.sy - p2.sx * p1.sy;
  }
  return a / 2;
}

  const R = 6371000;
  const toRad = d => d * Math.PI / 180;

// convex hull + rotating calipers would be ideal, but for speed/simplicity
// here we compute the bbox in a LOCAL projection aligned to the site's
// own principal axis (PCA-ish: try multiple rotation angles, keep best-fit
// rectangle with smallest area, since brownfield parcels are usually
// roughly axis-aligned to their own street frontage).
function localXY(ring) {
  const lats = ring.map(([lng,lat]) => lat);
  const lngs = ring.map(([lng,lat]) => lng);
  const lat0 = lats.reduce((a,b)=>a+b,0)/lats.length;

  const cos = Math.cos(angleRad), sin = Math.sin(angleRad);

function haversine(lat1, lng1, lat2, lng2) {
  const R = 6371000;
  const toRad = d => d * Math.PI / 180;
  const dLat = toRad(lat2-lat1), dLng = toRad(lng2-lng1);
  const a = Math.sin(dLat/2)**2 + Math.cos(toRad(lat1))*Math.cos(toRad(lat2))*Math.sin(dLng/2)**2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}
function localXY(ring) {
  const lats = ring.map(([lng,lat]) => lat);
  const lngs = ring.map(([lng,lat]) => lng);
  const lat0 = lats.reduce((a,b)=>a+b,0)/lats.length;
  const lng0 = lngs.reduce((a,b)=>a+b,0)/lngs.length;
  return ring.map(([lng,lat]) => {
    const x = haversine(lat0, lng0, lat0, lng) * (lng > lng0 ? 1 : -1);
    const y = haversine(lat0, lng0, lat, lng0) * (lat > lat0 ? 1 : -1);
    return [x, y];
  });
}
function rotatedBBoxArea(pts, angleRad) {
  const cos = Math.cos(angleRad), sin = Math.sin(angleRad);
  const rx = pts.map(([x,y]) => x*cos + y*sin);
  const ry = pts.map(([x,y]) => -x*sin + y*cos);
  const w = Math.max(...rx) - Math.min(...rx);
  const h = Math.max(...ry) - Math.min(...ry);
  return { w, h, area: w*h };
}
function minAreaBBox(pts) {
  let best = null;
  for (let deg = 0; deg < 180; deg += 2) {
    const r = rotatedBBoxArea(pts, deg * Math.PI / 180);
    if (!best || r.area < best.area) best = r;
  }
  return best;
}
