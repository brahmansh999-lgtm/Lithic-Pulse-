import { useMemo, useState } from 'react';
import { precedents, buildZoningSpec } from './zoningTargets';
import './ArchitectZoningGate.css';

// Step between SitePicker and the masterplan generator for the
// "reverse-engineer from a completed building" flow. The architect:
//   1. picks one of the 11 precedent buildings (extracted zoningSpec)
//   2. reviews/adjusts the scaled targets for THIS site
//   3. clicks Approve — nothing downstream (massing/columns/stairs) runs
//      until this happens. This is the manual authority checkpoint: the
//      AI proposes targets, the architect decides whether they're right
//      for the site before any 3D generation spends a single cycle.
export default function ArchitectZoningGate({ siteBounds, siteId, onApprove, onBack }) {
  const [pickedId, setPickedId] = useState(precedents[0]?.id ?? null);
  const [overrides, setOverrides] = useState({});

  const siteAreaM2 = (siteBounds?.widthM || 0) * (siteBounds?.depthM || 0);

  const picked = precedents.find(p => p.id === pickedId);
  const baseSpec = useMemo(
    () => (picked ? buildZoningSpec(picked, siteAreaM2) : null),
    [picked, siteAreaM2]
  );
  const spec = baseSpec ? { ...baseSpec, ...overrides } : null;

  const setField = (field, value) => setOverrides(o => ({ ...o, [field]: value }));

  return (
    <div className="azg-root">
      <div className="azg-header">
        {onBack && <button className="azg-back" onClick={onBack}>← Back</button>}
        <div className="azg-title">
          <h2>Zoning From Precedent</h2>
          <p>Pick a completed building as the reference. Its unit density, floor count and structural
             grid are scaled to this site — review and adjust before any massing is generated.</p>
        </div>
      </div>

      <div className="azg-body">
        <div className="azg-precedent-list">
          {precedents.map(p => (
            <button
              key={p.id}
              className={`azg-precedent-card ${pickedId === p.id ? 'azg-precedent-card--active' : ''}`}
              onClick={() => { setPickedId(p.id); setOverrides({}); }}
            >
              <div className="azg-precedent-name">{p.name}</div>
              <div className="azg-precedent-stats">
                <span>{p.floorCount ?? '—'} floors</span>
                <span>{p.coverageEstimatePct ?? '—'}% coverage</span>
                <span>{p.columnGridM ?? '—'}m grid</span>
              </div>
            </button>
          ))}
        </div>

        {spec && (
          <div className="azg-targets">
            <h3>Targets for this site</h3>
            <p className="azg-targets-sub">
              Scaled from {spec.sourceName} to a {(siteAreaM2 / 10000).toFixed(2)} ha site.
              Editable — your numbers override the scaled defaults.
            </p>

            <TargetRow
              label="Unit count"
              value={spec.targetUnitCount}
              onChange={v => setField('targetUnitCount', v)}
            />
            <TargetRow
              label="Floor count"
              value={spec.targetFloorCount}
              onChange={v => setField('targetFloorCount', v)}
            />
            <TargetRow
              label="Floor-to-floor (m)"
              value={spec.targetFloorToFloorM}
              step={0.1}
              onChange={v => setField('targetFloorToFloorM', v)}
            />
            <TargetRow
              label="Column grid (m)"
              value={spec.targetColumnGridM}
              step={0.1}
              onChange={v => setField('targetColumnGridM', v)}
            />
            <TargetRow
              label="Stair cores"
              value={spec.targetStairCoreCount}
              onChange={v => setField('targetStairCoreCount', v)}
            />

            <div className="azg-note">
              Green space is not set here — it's whatever ground is left once this density is matched
              on the site, reported on each variation rather than dialled in advance.
            </div>

            <button className="azg-approve" onClick={() => onApprove(spec)}>
              Approve Zoning →
            </button>
          </div>
        )}
      </div>
    </div>
  );
}

function TargetRow({ label, value, onChange, step = 1 }) {
  return (
    <div className="azg-target-row">
      <label>{label}</label>
      <input
        type="number"
        step={step}
        value={value ?? ''}
        onChange={e => onChange(e.target.value === '' ? null : Number(e.target.value))}
      />
    </div>
  );
}

import { useMemo, useState } from 'react';
import MasterplanViewer from './MasterplanViewer';
import { generatePrecedentFittedIterations } from './zoningTargets';
import './PrecedentVariations.css';

// Shown after the architect approves a zoningSpec (ArchitectZoningGate).
// Sweeps all 9 massing concepts against the approved targets and shows ONLY
// the ones that pass the zoning gate — labelled "Variation A/B/C...", never
// by concept name (courtyard_block, mountain_cascade, etc. are internal
// implementation detail, not a vocabulary the architect should have to
// reason in). A plan that doesn't meet the approved zoning never reaches
// this screen at all; it was filtered out before generation finished.
export default function PrecedentVariations({ siteBounds, sitePolygon, siteContext, analysis, zoningSpec, onOpenVariation, onBack }) {
  const [selected, setSelected] = useState(null);

  const baseOptions = useMemo(() => ({
    siteContext,
    solarAzimuth: analysis?.solarAzimuth,
    windAzimuth:  analysis?.windAzimuth,
    accessEdges:  analysis?.accessEdges,
  }), [siteContext, analysis]);

  const variations = useMemo(
    () => generatePrecedentFittedIterations(siteBounds, sitePolygon, zoningSpec, baseOptions, { keep: 12 }),
    [siteBounds, sitePolygon, zoningSpec, baseOptions]
  );

  return (
    <div className="pcv-root">
      <div className="pcv-header">
        {onBack && <button className="pcv-back" onClick={onBack}>← Back to Zoning</button>}
        <div className="pcv-title">
          <h2>Matching Variations</h2>
          <p>
            {variations.length
              ? `${variations.length} variation${variations.length === 1 ? '' : 's'} meet the approved zoning for ${zoningSpec.sourceName}.`
              : `No variation met the approved zoning for ${zoningSpec.sourceName} — try widening the targets.`}
          </p>
        </div>
      </div>

      <div className="pcv-grid">
        {variations.map(({ seed, plan, score, variationLabel }) => {
          const isSelected = seed === selected;
          return (
            <div
              key={seed}
              className={`pcv-card ${isSelected ? 'pcv-card--selected' : ''}`}
              onClick={() => setSelected(seed)}
            >
              <div className="pcv-card-thumb">
                <MasterplanViewer
                  plan={plan}
                  maxFloor={plan.floors.max}
                  stage="boxes"
                  selectedUnit={null}
                  onSelectUnit={() => {}}
                  filterTypology={null}
                  hoverTypology={null}
                  showSolar={false}
                />
              </div>
              <div className="pcv-card-meta">
                <span className="pcv-card-label">{variationLabel}</span>
                <span className="pcv-card-score">fit {score}</span>
              </div>
              <div className="pcv-card-stats">
                <span>{plan.totalUnits} homes</span>
                <span>{plan.greenPercent}% green</span>
                <span>{plan.floors.max} floors</span>
              </div>
              <button
                className="pcv-card-open"

                Open in editor →
              </button>
            </div>
          );
        })}
      </div>
    </div>
  );
}

// Derives column positions for a building's footprint, snapped to a
// centre-to-centre grid pitch (metres). This is NOT part of
// masterplanGenerator.js's massing/placement logic — it's a layer on top
// that reads a finished building's footprint + a target pitch (usually the
// approved precedent's targetColumnGridM from zoningTargets.js) and produces
// a real structural grid, the way an architect would overlay column lines on
// a massing study before any structural design begins.
//
// Why this exists as its own module: there was no column GEOMETRY anywhere
// in the masterplan pipeline before this — "columns" only existed as a
// per-unit material/BOM choice in the Configurator (computeBOM.js
// COLUMN_TYPES), entirely decoupled from plan position. Bringing precedent
// column data in meant generating real positions for the first time, not
// tuning an existing parameter.

import { CELL } from './masterplanGenerator';

// buildColumnGrid(building, gridPitchM) → [{ col, row }] in GRID CELLS
// (same coordinate space as building.col/row/footprint), one entry per
// column line intersection inside the building's footprint, inset half a
// bay from each edge so columns don't land exactly on the external wall.
export function buildColumnGrid(building, gridPitchM) {
  if (!gridPitchM || gridPitchM <= 0) return [];
  const pitchCells = gridPitchM / CELL;
  if (pitchCells < 1) return []; // pitch finer than the grid resolution — not meaningful at this scale

  const { col, row, w, d } = building;
  const columns = [];

  const inset = pitchCells / 2;
  for (let x = inset; x < w - inset * 0.5; x += pitchCells) {
    for (let z = inset; z < d - inset * 0.5; z += pitchCells) {
      columns.push({ col: col + Math.round(x), row: row + Math.round(z) });
    }
  }
  // De-duplicate (rounding can collapse two near-identical lines into the same cell on small footprints).
  const seen = new Set();
  return columns.filter(c => {
    const key = `${c.col},${c.row}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

// Attaches a columnGrid[] to every real building (units.length > 0) in a
// plan, using the approved zoning spec's targetColumnGridM. Call this once
// after generateMasterplan() / before rendering, on the chosen variation —
// not during the sweep, since the sweep only needs scoreAgainstZoning's
// numeric checks and computing full column grids for every swept candidate
// would be wasted work for the ones that get filtered out.
export function attachColumnGrids(plan, gridPitchM) {
  if (!plan || !gridPitchM) return plan;
  plan.buildings.forEach(b => {
    if (b.units?.length > 0) b.columnGrid = buildColumnGrid(b, gridPitchM);
  });
  return plan;
}

// Converts plan geometry into flat zone rectangles, brightest-first so later
// (dimmer) zones never visually overwrite a brighter one drawn before it —
// matches the social-importance stacking order (lobby paints last/on top).
function buildZones(plan) {
  if (!plan) return [];
  const zones = [];

  // Dimmest first: green/open ground as the base layer (whole buildable area
  // minus everything else gets implicitly shown via the white background —
  // explicit green zones from deckGardens still get their own swatch).
  (plan.deckGardens || []).forEach(g => zones.push({ kind: 'green', col: g.col, row: g.row, w: 1, d: 1 }));

  // Units/private — every real building footprint.
  (plan.buildings || []).forEach(b => {
    if (!b.units?.length) return;
    zones.push({ kind: 'unit', col: b.col, row: b.row, w: b.w, d: b.d });
  });

  // Circulation — stair cores + exterior stairs.
  (plan.buildings || []).forEach(b => {
    (b.stairCores || []).forEach(c => zones.push({ kind: 'circulation', col: c.col, row: c.row, w: 1, d: 1 }));
  });
  (plan.exteriorStairs || []).forEach(s => zones.push({ kind: 'circulation', col: s.col, row: s.row, w: 1, d: 1 }));

  // Community/amenity — pavilions (communal program on the podium deck).
  (plan.pavilions || []).forEach(p => zones.push({ kind: 'community', col: p.col, row: p.row, w: p.w || 2, d: p.d || 2 }));

  // Brightest: lobbies + entry points.
  (plan.buildings || []).forEach(b => {
    (b.lobbies || []).forEach(l => zones.push({ kind: 'lobby', col: l.col, row: l.row, w: l.w || 2, d: 1 }));
  });
  (plan.roads || []).filter(r => r.type === 'entry').forEach(r => zones.push({ kind: 'lobby', col: r.col, row: r.row, w: 1, d: 1 }));

  return zones;
}

// Fetches REAL surrounding building footprints + road centrelines around a
// site from the public Overpass API (OpenStreetMap), and converts them into
// the same normalised 0..1 x/z grid space as sitePolygon — so they can be
// rendered in MasterplanViewer as flat pale context massing + real road
// lines around the designed plan, instead of either nothing or fake
// procedural geometry.
//
// This is genuinely live network data: no offline fallback dataset exists in
// this app for neighbouring buildings (brownfield_london.geojson only has
// the curated DEVELOPMENT sites themselves, not their surroundings). A
// failed/slow fetch must never block the masterplan from rendering — it's
// decorative context, so errors resolve to empty arrays, not thrown.

import { useEffect, useState } from 'react';

const OVERPASS_ENDPOINT = 'https://overpass-api.de/api/interpreter';

// How far around the site to pull context, in degrees — ~180m at London's
// latitude, enough to read "the site sits in a city block" without pulling
// in an expensive wide-area query.
const CONTEXT_PAD_DEG = 0.0016;

function buildQuery(minLat, minLng, maxLat, maxLng) {
  const bbox = `${minLat - CONTEXT_PAD_DEG},${minLng - CONTEXT_PAD_DEG},${maxLat + CONTEXT_PAD_DEG},${maxLng + CONTEXT_PAD_DEG}`;
  return `
    [out:json][timeout:15];
    (
      way["building"](${bbox});
      way["highway"](${bbox});
    );
    out geom;
  `;
}

// Converts a lat/lng using the SAME basis SitePicker.jsx computed for the
// site polygon itself (geoRef), so OSM geometry lands in identical grid
// space to sitePolygon/plan.sitePolyGrid — no separate re-projection logic.
function toNormXZ(lat, lng, geoRef) {
  return {
    x: ((lng - geoRef.minLng) * geoRef.lngScale) / geoRef.rLng,
    z: (lat - geoRef.minLat) / geoRef.rLat,
  };
}

// useOsmContext(geoRef) → { buildings: [{points:[{x,z}], levels}], roads: [{points:[{x,z}]}], loading }
// buildings/roads coordinates are in the SAME normalised space as
// sitePolygon — multiply by cols*CELL / rows*CELL exactly like SiteFloor
// does to place them in world space.
export function useOsmContext(geoRef) {
  const [data, setData] = useState({ buildings: [], roads: [], loading: false });

  useEffect(() => {
    if (!geoRef) { setData({ buildings: [], roads: [], loading: false }); return; }
    const controller = new AbortController();
    setData(d => ({ ...d, loading: true }));

    const query = buildQuery(geoRef.minLat, geoRef.minLng, geoRef.maxLat, geoRef.maxLng);
    fetch(OVERPASS_ENDPOINT, { method: 'POST', body: query, signal: controller.signal })
      .then(r => { if (!r.ok) throw new Error(`Overpass ${r.status}`); return r.json(); })
      .then(json => {
        const buildings = [];
        const roads = [];
        for (const el of json.elements || []) {
          if (!el.geometry || el.geometry.length < 2) continue;
          const points = el.geometry
            .filter(g => g && g.lat != null && g.lon != null)
            .map(g => toNormXZ(g.lat, g.lon, geoRef));
          if (points.length < 2) continue;

          if (el.tags?.building) {
            if (points.length >= 3) {
              const levels = parseInt(el.tags['building:levels'], 10);
              buildings.push({ points, levels: Number.isFinite(levels) ? levels : null });
            }
          } else if (el.tags?.highway) {
            roads.push({ points });
          }
        }
        setData({ buildings, roads, loading: false });
      })
      .catch(() => {
        if (!controller.signal.aborted) setData({ buildings: [], roads: [], loading: false });
      });

    return () => controller.abort();
  }, [geoRef]);

  return data;
}
